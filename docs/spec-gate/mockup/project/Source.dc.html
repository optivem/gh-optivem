public class RefundService {

    public Refund refund(OrderId id) {
        var order = orders.get(id);
        if (order.deliveredMoreThanDaysAgo(30)) {
            throw new RefundNotAllowedException(id);
        }
        return new Refund(order.total());
    }

    public Refund refund(OrderId id, List<ItemId> items) {
        throw new UnsupportedOperationException(
            "partial refund not implemented");
    }
}
